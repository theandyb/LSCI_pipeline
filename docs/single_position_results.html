get_pos_results <- function(res_dir, pop, subtype){
  f_name <- paste0(res_dir, "/", pop, "/", subtype, ".csv")
  df <- read_csv(f_name, show_col_types = FALSE) %>%
    rowwise() %>%
    mutate(re = dev / (2 * (singletons+controls)))
  return(df)
}

get_pos_cc <- function(control_control, pop, subtype){
  f_name <- paste0(control_control, "/", pop, "/", subtype, "_v2.csv")
  df <- read_csv(f_name, show_col_types = FALSE) %>%
           mutate(re = dev / (2 * (singletons + controls) ))
  return(df)
}

get_pos_results_all <- function(res_dir, pop, subtypes){
  results <- vector(mode = "list", length = length(subtypes))
  
  for(i in 1:length(subtypes)){
    results[[i]] <- get_pos_results(res_dir, pop, subtypes[[i]])
    results[[i]]$subtype <- subtypes[[i]]
  }
  
  return(bind_rows(results))
}

subtype_print_names <- function(st){
  if(str_starts(st, "AT")){
    return(paste0("A → ", str_sub(st, 4, 4)))
  } else if(str_starts(st, "GC")){
    return(paste0("C → ", str_sub(st, 5, 5)))
  } else if(str_starts(st, "all")){
    return(paste0("C → ", str_sub(st, 9, 9)))
  } else{
    return(paste0("CpG → ", str_sub(st, 9, 9), "pG"))
  }
}

get_pos_resid <- function(res_dir, pop, subtype, rp){
  f_name <- paste0(res_dir, "/resid/", pop, "/", subtype, "_rp_", rp, ".csv")
  df <- read_csv(f_name, show_col_types = FALSE)
  df$rp <- rp
  
  n_tot <- sum(df$n)
  df$re_res <- (df$res^2) / (2*n_tot)
  return(df)
}

get_pos_resid_all <- function(res_dir, pop, subtype, rp = c(-500:-1, 1:500)){
  results <- vector(mode = "list", length = length(rp))
  for(i in 1:length(rp)){
    if(rp[i] == 1 & str_starts(subtype, "cpg")) next
    results[[i]] <- get_pos_resid(res_dir, pop, subtype, rp[i])
  }
  df <- bind_rows(results)
  return(df)
}

    select(ID, nuc, status, res) %>%

    rowwise() %>%
    mutate(dev_cont = sign(singletons) * (singletons ^ 2 + controls ^ 2)) %>%

Text

Our analyses investigated the influence of neighboring nucleotides on germline mutation rates. We assessed the influence of nucleotides at individual flanking positions in an expanded window covering flanking positions up to 1000 bp from the focal position. Second and third order interactions were assessed across all combinations of flanking positions within +/- 10 bp of the focal position to ascertain the degree to which neighboring nucleotides interact and moderate each other’s influence on the rate of mutation. We conducted these analyses using singletons aggregated across all 1000 Genomes Project (1kGP) populations and separately for each of the five 1kGP super-populations. We then assessed the consistency of population-level results for both single-position and higher-order models.

Patterns of influence at individual flanking positions in an expanded window

We first examined the influence of nucleotides at individual flanking positions within a small window (e.g., +/- 10bp from the focal position), consistent with previous research on local sequence context influence on germline mutation rates (citation). However we considered nine distinct mutation categories, as we addressed the known impact of CpG dinucleotides on mutation rates by separately evaluating CpG and non-CpG mutations as distinct categories. Across all nine mutation types, we observed highly statistically significant deviance statistics at all flanking positions within 10 bases of the focal position, ranging from 18.6 to 1659567 (\(\chi^2\) df = 3, \(p \leq 3.24 \times 10^-4\)).
